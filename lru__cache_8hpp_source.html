59 if (capacity_ == 0) {

61 }
62 std::lock_guard<std::mutex> guard(mutex_);

64 if (it != index_.end()) {
65 it->second->second = std::move(value);

69 items_.emplace_front(key, std::move(value));
70 index_.emplace(key, items_.begin());

76 }
